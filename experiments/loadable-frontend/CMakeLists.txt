# Matching-build proof, not a production promise of cross-build ABI stability.
set(ABI_INPUT "${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CMAKE_SIZEOF_VOID_P};${CMAKE_BUILD_TYPE};WINDOWS;MD")
file(GLOB_RECURSE ABI_HEADERS "${ENGINE}/src/*.h")
list(APPEND ABI_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/export-api.h" "${CMAKE_CURRENT_SOURCE_DIR}/frontend-stream.h")
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    list(APPEND ABI_HEADERS ${CORE_OBJECTS})
else()
    list(APPEND ABI_HEADERS "${ENGINE_BUILD}/anybandui_engine.lib")
endif()
foreach(input IN LISTS ABI_HEADERS)
    file(SHA256 "${input}" digest)
    string(APPEND ABI_INPUT "${digest}")
endforeach()
string(SHA256 ABI_ID "${ABI_INPUT}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/engine-api.h" TABLE_DIGEST)
string(SHA256 TABLE_ID "${ABI_ID};${TABLE_DIGEST}")
file(WRITE "${CMAKE_BINARY_DIR}/build-id.h" "#define FRONTEND_BUILD_ID \"${ABI_ID}\"\n#define FRONTEND_TABLE_ID \"${TABLE_ID}\"\n")
include_directories(BEFORE "${CMAKE_BINARY_DIR}")
add_compile_definitions(WINDOWS _CRT_SECURE_NO_WARNINGS)
add_compile_options(/std:clatest /we4013 /we4047 /we4024 /utf-8)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}" "${ENGINE}/src" "${ADAPTER}/vendor/cjson")
add_library(frontend MODULE plugin.c)
target_link_libraries(frontend PRIVATE ${JSON_LINK})

    ${ENGINE_LINK} ${JSON_LINK})

target_link_libraries(frontend-loader-test PRIVATE frontend_host ${ENGINE_LINK})
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set_target_properties(frontend-session frontend-loader-test PROPERTIES EXCLUDE_FROM_ALL TRUE)

    target_include_directories(angband-plugin PRIVATE "${ENGINE}/src/win/include")
    target_link_libraries(angband-plugin PRIVATE frontend_host ${ENGINE_LINK} ${JSON_LINK}

    target_link_libraries(frontend-export-loader-test PRIVATE frontend_exports_host)
