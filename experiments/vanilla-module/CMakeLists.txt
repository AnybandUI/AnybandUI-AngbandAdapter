cmake_minimum_required(VERSION 3.22)
project(anybandui_vanilla_module C)
set(ENGINE_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../../build/vanilla-modules" CACHE PATH "Matching Angband build")
set(ENGINE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../angband" CACHE PATH "Engine headers")
file(GLOB PRESENTATION CONFIGURE_DEPENDS presentation/*.c)
add_library(anybandui MODULE ${PRESENTATION} src/main-anybandui.c document.c module.c
    ../../vendor/cjson/cJSON.c)
target_include_directories(anybandui PRIVATE sdk src ../../vendor/cjson "${ENGINE_BUILD}")
target_compile_definitions(anybandui PRIVATE WINDOWS _CRT_SECURE_NO_WARNINGS)
target_compile_options(anybandui PRIVATE "$<$<COMPILE_LANGUAGE:C>:/std:clatest;/we4013;/we4047;/we4024>" /utf-8)
target_link_libraries(anybandui PRIVATE "${ENGINE_BUILD}/angband.lib")

option(FRONTEND_SESSION_CHECK "Include in-process semantic regression checks" OFF)
if(FRONTEND_SESSION_CHECK)
    target_sources(anybandui PRIVATE tests/session.c)
    target_compile_definitions(anybandui PRIVATE FRONTEND_SESSION_CHECK)
    set_source_files_properties(tests/session.c PROPERTIES COMPILE_DEFINITIONS main=frontend_session_test)
    add_library(frontend-wrong-build MODULE tests/wrong-module.c)
    target_include_directories(frontend-wrong-build PRIVATE sdk)
endif()
# The checked-in SDK is pinned. Do not bless stale headers with a new host ID.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk/frontend-build.h" SDK_ID)
file(READ "${ENGINE_BUILD}/frontend-build.h" HOST_ID)
if(NOT SDK_ID STREQUAL HOST_ID)
    message(FATAL_ERROR "This adapter SDK does not match the selected engine build. Port and regenerate the SDK first.")
endif()
set(ANYBANDUI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../AnybandUI" CACHE PATH "Actual client source")
option(FRONTEND_CLIENT_CHECK "Include the real AnybandUI connection scenario" OFF)
if(FRONTEND_CLIENT_CHECK)
    enable_language(CXX)
    set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
    set(FETCHCONTENT_SOURCE_DIR_SDL3 "${ANYBANDUI_SOURCE}/build-anybandui/_deps/sdl3-src" CACHE PATH "")
    set(FETCHCONTENT_SOURCE_DIR_IMGUI "${ANYBANDUI_SOURCE}/build-anybandui/_deps/imgui-src" CACHE PATH "")
    set(FETCHCONTENT_SOURCE_DIR_JSON "${ANYBANDUI_SOURCE}/build-anybandui/_deps/json-src" CACHE PATH "")
    file(READ "${ANYBANDUI_SOURCE}/protocol/full-v1.json" ANYBANDUI_CONTRACT_JSON)
    configure_file("${ANYBANDUI_SOURCE}/protocol/protocol_contract.h.in" generated/protocol_contract.h @ONLY)
    add_subdirectory("${ANYBANDUI_SOURCE}" client EXCLUDE_FROM_ALL)
    target_sources(anybandui PRIVATE client-scenario.cpp)
    target_compile_features(anybandui PRIVATE cxx_std_17)
    target_compile_definitions(anybandui PRIVATE FRONTEND_CLIENT_CHECK)
    target_include_directories(anybandui PRIVATE "${ANYBANDUI_SOURCE}/anybandui")
    target_link_libraries(anybandui PRIVATE anybandui_crt)
endif()
